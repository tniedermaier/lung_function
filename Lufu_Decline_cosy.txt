# Vorbereitung für Table 2:
lufu_random_slopes <- function(data, FUN=summary){
 dat_long <- tidyr::pivot_longer(data, cols=c("FEV1_GLI_V1","FEV1_GLI_V3","FEV1_GLI_V4","FEV1_GLI_V5","FEV1_GLI_V6"), names_to="FEV1_GLI")
 class(dat_long) <- "data.frame"
 colnames(dat_long)[colnames(dat_long)=="value"] <- "FEV1"
 dat_long$visit <- as.numeric(substr(dat_long$FEV1_GLI, 11, 11))

 dat_long$time3 <- NA
 dat_long[dat_long$visit==1,]$time3 <- 0
 #dat_long[dat_long$visit==2,]$time3 <- 6
 dat_long[dat_long$visit==3,]$time3 <- (dat_long[dat_long$visit==3,]$DATUM_V3 - dat_long[dat_long$visit==3,]$DATUM_V1) / 365.25
 dat_long[dat_long$visit==4,]$time3 <- (dat_long[dat_long$visit==4,]$DATUM_V4 - dat_long[dat_long$visit==4,]$DATUM_V1) / 365.25
 dat_long[dat_long$visit==5,]$time3 <- (dat_long[dat_long$visit==5,]$DATUM_V5 - dat_long[dat_long$visit==5,]$DATUM_V1) / 365.25
 dat_long[dat_long$visit==6,]$time3 <- (dat_long[dat_long$visit==6,]$DATUM_V6 - dat_long[dat_long$visit==6,]$DATUM_V1) / 365.25

 dat_long <- dat_long[complete.cases(dat_long$FEV1),]
 dat_long <<- dat_long[,c("FEV1","time3","subject")] #beschränken auf relevante Variablen
 lmer_random_slope <<- lmer(FEV1 ~ time3 + (1+time3|subject)# + dropout_reason_name
 , data=dat_long, control = lmerControl(optimizer ="Nelder_Mead"))
 ann_declines <<- fixef(lmer_random_slope)[2] + ranef(lmer_random_slope)$subject[,2]
 IDs <<- rownames(ranef(lmer_random_slope)$subject)
 intercept <<- fixef(lmer_random_slope)[1]
 slope <<- FUN(fixef(lmer_random_slope)[2] + ranef(lmer_random_slope)$subject[,2])["Mean"]
 FUN(fixef(lmer_random_slope)[2] + ranef(lmer_random_slope)$subject[,2])
}

#### Mean Slope (model)^1 [original values / not winsorized]:

lufu_random_slopes_big_model <- function(data, FUN=summary){
 dat_long <- tidyr::pivot_longer(data, cols=c("FEV1_GLI_V1","FEV1_GLI_V3","FEV1_GLI_V4","FEV1_GLI_V5","FEV1_GLI_V6"), names_to="FEV1_GLI")
 class(dat_long) <- "data.frame"
 colnames(dat_long)[colnames(dat_long)=="value"] <- "FEV1"
 dat_long$visit <- as.numeric(substr(dat_long$FEV1_GLI, 11, 11))

 dat_long$time3 <- NA
 dat_long[dat_long$visit==1,]$time3 <- 0
 #dat_long[dat_long$visit==2,]$time3 <- 6
 dat_long[dat_long$visit==3,]$time3 <- (dat_long[dat_long$visit==3,]$DATUM_V3 - dat_long[dat_long$visit==3,]$DATUM_V1) / 365.25
 dat_long[dat_long$visit==4,]$time3 <- (dat_long[dat_long$visit==4,]$DATUM_V4 - dat_long[dat_long$visit==4,]$DATUM_V1) / 365.25
 dat_long[dat_long$visit==5,]$time3 <- (dat_long[dat_long$visit==5,]$DATUM_V5 - dat_long[dat_long$visit==5,]$DATUM_V1) / 365.25
 dat_long[dat_long$visit==6,]$time3 <- (dat_long[dat_long$visit==6,]$DATUM_V6 - dat_long[dat_long$visit==6,]$DATUM_V1) / 365.25

 dat_long <- dat_long[complete.cases(dat_long$FEV1),]
 dat_long <<- dat_long[,c("FEV1","time3","subject","dropout_reason_name","GOLD_GLI_V2")] #beschränken auf relevante Variablen
 (lmer_res <<- lmer(FEV1 ~ time3*dropout_reason_name*as.factor(GOLD_GLI_V2) + (1+time3|subject), data=dat_long, control = lmerControl(optimizer ="Nelder_Mead"))) #Hier mit GOLD_GLI_V2, wie es sein soll!
 
 ann_declines <<- fixef(lmer_res)[2] + ranef(lmer_res)$subject[,2]
 IDs <<- rownames(fixef(lmer_res)[2] + ranef(lmer_res)$subject)
 return(lmer_res)
}

lufu_random_slopes_ml_rtm2 <- function(data, FUN=summary, mytrim=0){
 dat_long <- tidyr::pivot_longer(data, cols=c("LUFU_FEV1_V1","LUFU_FEV1_V3","LUFU_FEV1_V4","LUFU_FEV1_V5","LUFU_FEV1_V6")) #Hier FEV1_GLI_V1 weggelassen, stattdessen ..._V2 gewählt.
 class(dat_long) <- "data.frame"
 colnames(dat_long)[colnames(dat_long)=="value"] <- "FEV1"
 dat_long$visit <- as.numeric(substr(dat_long$name, 12, 12))

 dat_long$time3 <- NA
 dat_long[dat_long$visit==1,]$time3 <- 0
 dat_long[dat_long$visit==3,]$time3 <- (dat_long[dat_long$visit==3,]$DATUM_V3 - dat_long[dat_long$visit==3,]$DATUM_V1) / 365.25
 dat_long[dat_long$visit==4,]$time3 <- (dat_long[dat_long$visit==4,]$DATUM_V4 - dat_long[dat_long$visit==4,]$DATUM_V1) / 365.25
 dat_long[dat_long$visit==5,]$time3 <- (dat_long[dat_long$visit==5,]$DATUM_V5 - dat_long[dat_long$visit==5,]$DATUM_V1) / 365.25
 dat_long[dat_long$visit==6,]$time3 <- (dat_long[dat_long$visit==6,]$DATUM_V6 - dat_long[dat_long$visit==6,]$DATUM_V1) / 365.25

 dat_long <- dat_long[complete.cases(dat_long$FEV1),]
 dat_long <<- dat_long[,c("FEV1","time3","subject")] #beschränken auf relevante Variablen
 lmer_random_slope <- lmer(FEV1 ~ time3 + (1+time3|subject)# + dropout_reason_name
 , data=dat_long, control = lmerControl(optimizer ="Nelder_Mead"))
 ann_declines <<- fixef(lmer_random_slope)[2] + ranef(lmer_random_slope)$subject[,2]
 IDs <<- rownames(ranef(lmer_random_slope)$subject)
 intercept <<- fixef(lmer_random_slope)[1]
 FUN(psych::winsor(1000*(fixef(lmer_random_slope)[2] + ranef(lmer_random_slope)$subject[,2]), trim=mytrim))
}
